IMM 2025-26 \quad Complex Analysis \quad Exercises 5.   \hfill  
\smallskip
\hrule
\bn
\item{1.}  Evaluate $\Gamma(-{1\over 2}),\quad \Gamma(-{7\over  2}),\quad \Gamma(-{1\over 3})$. 

 
\bn
\item{2.}   Let $n\in\N$. Show that 
$$ \Gamma\left({1\over 2}+n\right)\,=\, {{1\cdot 3\cdot 5\cdot \ldots \cdot (2n-1)}\over {2^n}}\sqrt\pi\, 
%=\, 2^{1-2n}{{\Gamma(2n)}\over {\Gamma(n)}}\sqrt\pi
.$$ 
 
\bn
\item{3.}   Use  Wielandt's theorem to prove that for $Re(z)>0$ 
$$\Gamma(z)={1\over {g(z)}},\quad  
 \hbox{where}\quad g(z)=ze^{Cz}\prod_{k=1}^\infty\left((1+{z\over k})e^{-z/k}\right),$$
where $C=\lim_k(1+{1\over 2}+{1\over 3}+\ldots-\log k)$ is the Euler-Mascheroni constant. \par
 ({\it The definition of $\Gamma$ as the  reciprocal of the above function $g$ is in Cartan's book, p.159-161}).



\bn
\item{4.} Use  Wielandt's theorem to prove the Legendre duplication formula 
$$\Gamma(2z)={1\over {\sqrt\pi}} 2^{2z-1}\Gamma(z)\Gamma(z+{1\over 2}).$$
 (hint: let $w=2z$ and consider  the function $f(w)=2^w\Gamma({w\over 2})\Gamma({w\over 2}+{1\over 2})$). 

 
\bn
\item{5.} Prove that if $y>0$, then 
$$|\Gamma(iy)|=\sqrt{{\pi\over{y\sinh \pi y}}}.$$ 

\bn\bn\bn
\item{6.}  Recall that  $\zeta(s)=\prod_{p}{1\over{1-{1\over p^s}}},$ for $s\in\C$ with $Re(s)>1$. Compute the logarithmic derivative of $\zeta$ (justify the steps). 

\bn
\item{7.}   Let $2,\,3,\,5,\,7,\,\ldots$ be the series of prime numbers. Prove that 
$$(1-{1\over 2^2})(1-{1\over 3^2})(1-{1\over 5^2})(1-{1\over 7^2})\ldots={6\over{\pi^2}}.$$ 
 
\bn
\item{8.}  Verify that $\zeta(1-s)=2^{1-s}\pi^{-s}\Gamma(s) \cos({\pi s\over  2})\zeta(s)$. 

\bn
\item{9.}  Using the analytic continuation given by the   formula of the previous exercise, prove that $\zeta(-1)=-{1\over 12}$  and that $\zeta(-3)= {1\over 120}$. 

\bn
\item{10.}  Compute $\zeta(m)$, where $m$ is a negative  integer.
